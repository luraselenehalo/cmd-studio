  ```powershell
  $run = { Invoke-GuestDesktop $vm "$g\recovery-run.ps1" "-Show $dataArg" $u }
  $appLog = { Invoke-GuestPs $vm "Select-String -LiteralPath '$data\logs\cmdstudio.log' -Pattern ' run |recovery_entry|renderer |pause reason|resume by|crash_check|cmd=app_|cmd=crash_|cmd=settings_reset' | ForEach-Object Line | Select-Object -Last 20; '--- the last 6 lines'; Get-Content -Tail 6 -Encoding UTF8 -LiteralPath '$data\logs\cmdstudio.log'" $u }

  $exe = 'C:\cmdstudio\phase3\app\cmdstudio.exe'   # the artifact's exe, as start-app.vbs names it (Install-Phase3App copies it there)
  ```
